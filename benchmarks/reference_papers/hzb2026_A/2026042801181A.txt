def speed(self, minute: float) -> float:
"""返回当前分钟对应的平均车速。

求解中采用均值：
顺畅 55.3 km/h，一般 35.4 km/h，拥堵 9.8

t = minute % 1440

# 若 Excel 只给出了 8:00-17:00 的速度，
则其他时间默认按一般时段处理。
return 35.4
def period_end(self, minute: float) -> float:
"""返回当前速度时段的结束分钟，用于跨
时段分段行驶计算。"""
t = minute % 1440
for p in self.speeds:

if p["start"] <= t < p["end"]:
return minute + (p["end"] - t)
future_starts = [p["start"] for p in self.speeds
if p["start"] > t]

def travel_segment(self, km: float, start_minute:
float, v: Vehicle, load_ratio: float) -> Tuple[float, float,
float, float]:
"""计算一段路程的分段行驶结果。
返回：
new_time: 到达时间，单位分钟；
energy_cost: 能源成本，单位元；
emission: 碳排放量，单位 kg；
carbon_cost: 碳排放成本，单位元。
这样处理后，如果车辆从拥堵时段开到顺
畅时段，程序会自动按两个速度段分别计算，
更符合数学建模文档中“跨越时间段时采
用分段距离计算”的要求。
"""
left_km = max(0.0, float(km))
t = float(start_minute)
energy_cost = 0.0
emission = 0.0
carbon_cost = 0.0
guard = 0
while left_km > 1e-9 and guard < 20:
guard += 1
sp = self.speed(t)
end_t = self.period_end(t)
available_min = max(1e-6, end_t - t)
max_km_this_period = sp *
available_min / 60.0
use_km = min(left_km,
max_km_this_period)
e, co2, cc = self.energy(use_km, sp, v,
load_ratio)
energy_cost += e
emission += co2
carbon_cost += cc
t += use_km / sp * 60.0

def must_use_electric(self, c: Customer) -> bool:
"""第二问核心规则：绿色配送区内，且时

视为必须由新能源车服务。

def allowed(self, v: Vehicle, c: Customer) -> bool:
"""判断车辆 v 是否允许服务客户 c。
第一问：所有车型均可服务所有客户。
第二问：若客户属于绿色配送区，且服务
时间窗与 8:00-16:00 限行时段相交，
则燃油车不能服务该客户，只能由新能源
车服务。
"""
if self.cfg.policy.lower() != "q2":
return True
if self.must_use_electric(c) and v.energy ==
"fuel":

def vehicle_pool(self) -> List[Vehicle]:
priority = {"EV-Heavy": 1, "Fuel-Heavy": 2,
"EV-Light": 3, "Fuel-Medium": 4, "Fuel-Light": 5}
pool = []
for v in sorted(self.vehicles, key=lambda x:

@staticmethod
def take(c: Customer, rw: float, rv: float) ->
Optional[Stop]:
if rw <= 1e-9 or rv <= 1e-9 or c.remain_w
<= 1e-9 or c.remain_v <= 1e-9:

c.remain_v)

c.remain_w -= w; c.remain_v -= v
if c.remain_w < 1e-6: c.remain_w = 0.0
if c.remain_v < 1e-6: c.remain_v = 0.0
return Stop(c.node_id, c.x, c.y, w, v,
c.tw_start, c.tw_end, c.green)
def build_route(self, idx: int, v: Vehicle) ->
Optional[Route]:
"""构造单辆车路径。
第二问优化重点：
1. 新能源车优先装载“必须由新能源车服
务”的绿色区客户；
2. 燃油车自动跳过绿色区限行客户；
3. 每辆车按容量动态分担客户需求，允许
大客户被多辆车分批服务。
"""
left = self.left()
# 若仍存在必须由新能源服务的绿色区客
户，新能源车优先处理这些客户，
# 避免新能源车被普通客户提前占用，导
致绿色区客户未服务。
must_ev_left = [c for c in left if
self.must_use_electric(c)]
if self.cfg.policy.lower() == "q2" and
v.energy != "fuel" and must_ev_left:
feasible = must_ev_left
else:
feasible = [c for c in left if
self.allowed(v, c)]
if not feasible:
return None
# 远端客户优先作为种子，减少远距离客
户单独派车的概率。
seed = max(feasible, key=lambda c:
self.cost.d(0, c.node_id))
rw, rv, stops, visited = v.weight_cap,
v.volume_cap, [], set()
first = self.take(seed, rw, rv)
if first is None:

===== PAGE 41 =====
return None
stops.append(first); visited.add(first.node_id)
rw -= first.weight; rv -= first.volume
while rw > 50 and rv > 0.02:
left_now = self.left()
must_ev_now = [c for c in left_now if
self.must_use_electric(c)]
if self.cfg.policy.lower() == "q2" and
v.energy != "fuel" and must_ev_now:
# 只要还有绿色限行客户没服务
完，新能源车继续优先装绿色限行客户。
candidates = [c for c in
must_ev_now if c.node_id not in visited]
else:
candidates = [
c for c in left_now
if c.node_id not in visited and
self.allowed(v, c)
]
if not candidates:
break
last = stops[-1]
# 最近邻填充，使路径更短。
nxt = min(candidates, key=lambda c:
self.cost.d(last.node_id, c.node_id))
stop = self.take(nxt, rw, rv)
if stop is None:
break
stops.append(stop);
visited.add(stop.node_id)
rw -= stop.weight; rv -= stop.volume
if rw / v.weight_cap < 0.015 or rv /
v.volume_cap < 0.015:
break
return self.cost.best_order(idx, v, stops)
def solve(self) -> Tuple[List[Route],
List[Customer]]:

for v in self.vehicle_pool():
if not self.left():
break
route = self.build_route(idx, v)
if route:
self.routes.append(route)
idx += 1
return self.routes, self.left()

for vehicle in vehicles:

def cvar_late_penalty(self, vehicle: Vehicle,
start_node: int, stops: List[Stop],
start_time: float,

"""蒙特卡洛模拟时变速度，计算迟到罚金
CVaR。"""

vehicle, stops, start_time,
start_node=start_node,
include_fixed=False,
random_mode=True
)
penalties.append(float(m["late_cost"]))

0.0
if len(arr) == 0:
return expected, 0.0, penalties
threshold = np.quantile(arr,
self.cfg.risk_alpha)
tail = arr[arr >= threshold]
cvar = float(tail.mean()) if len(tail) else
expected
return expected, cvar, penalties
